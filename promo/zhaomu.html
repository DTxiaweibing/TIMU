1. 架构全景

核心链路：用户提问 → 本地嵌入（ONNX）→ 本地向量检索（sqlite-vec）→ 上下文拼装 → [生成层接口]

用户提问
    ↓
嵌入模型（22.83MB，BGE-small-zh INT8）
    ↓
sqlite-vec 向量检索
    ↓
拼装 Prompt（上下文裁剪）
    ↓
生成模型选择器
    ├── 默认：云端 GLM-4-Flash（极速，免费额度大）→ 流式回答
    └── 断网/离线：本地 Qwen-0.5B（懒加载 400MB）→ 流式回答
关键设计铁律：
  • 生成层依赖注入，云端与本地实现同一个 TextGenerator 接口，切换成本为零。
  • 检索与生成完全解耦，替换生成模型不影响检索链路。

3. 阶段一：知识库预处理（PC端）

3.1 文本分块规则

  • 每块最大 256 tokens（约 180~200 个中文字符）
  • 相邻块 32 tokens 重叠（约 12.5%）
  • 句边界感知（不在句子中间断开）

Python 分块代码：

import re

def smart_chunk(text, max_tokens=256, overlap_tokens=32):
    sentences = re.split(r'(?<=[。！？.!?])', text)
    chunks = []
    current = []
    current_len = 0
    for sent in sentences:
        sent_len = len(sent.encode('utf-8')) // 3
        if current_len + sent_len > max_tokens and current:
            chunks.append(''.join(current))
            overlap = current[-1][-overlap_tokens*3:] if current else ''
            current = [overlap, sent] if overlap else [sent]
            current_len = len(overlap.encode('utf-8'))//3 + sent_len
        else:
            current.append(sent)
            current_len += sent_len
    if current:
        chunks.append(''.join(current))
    return chunks

3.2 生成向量并写入 sqlite-vec 数据库

import sqlite3, json, numpy as np
from sentence_transformers import SentenceTransformer

# 加载 BGE 模型（PC端用 PyTorch 版）
model = SentenceTransformer('BAAI/bge-small-zh-v1.5')

# 分块
chunks = smart_chunk(open('knowledge.txt', 'r', encoding='utf-8').read())

# 生成向量（BGE 默认已做 L2 归一化）
vectors = model.encode(chunks, normalize_embeddings=True)

# 转为 float16 存储（体积减半，精度损失极小）
vectors_fp16 = vectors.astype(np.float16)

conn = sqlite3.connect('knowledge.db')

# 创建向量表（维度固定为 512！）
conn.execute("CREATE VIRTUAL TABLE kb_vec USING vec0(embedding FLOAT[512])")

# 插入向量
for i, vec in enumerate(vectors_fp16):
    conn.execute("INSERT INTO kb_vec(rowid, embedding) VALUES (?, ?)",
                 (i, vec.tobytes()))

# 创建文本映射表
conn.execute("CREATE TABLE chunk_text(id INTEGER, text TEXT)")
for i, chunk in enumerate(chunks):
    conn.execute("INSERT INTO chunk_text VALUES (?, ?)", (i, chunk))

# 【关键】写入版本号（用于 App 端热更新校验）
conn.execute("PRAGMA user_version = 2")  # 每次知识库变更手动 +1

conn.commit()
conn.close()

print(f"✅ 知识库构建完成，共 {len(chunks)} 个分块，版本号: 2")

5.1 BertTokenizer（读取 vocab.txt）

class BertTokenizer(private val vocab: Map) {
    private val unkId = 100   // [UNK] 的 ID
    private val clsId = 101   // [CLS]
    private val sepId = 102   // [SEP]
    private val maxLen = 512

    companion object {
        fun fromAssets(context: Context): BertTokenizer {
            val vocab = context.assets.open("models/vocab.txt").bufferedReader().useLines { lines ->
                lines.mapIndexed { index, word -> word.trim() to index }.toMap()
            }
            return BertTokenizer(vocab)
        }
    }

    fun encode(text: String): IntArray {
        val tokens = mutableListOf(clsId)
        val rawTokens = text.lowercase().trim().split(Regex("""(?<=\p{P})|(?=\p{P})|\s+"""))
        for (token in rawTokens) {
            if (token.isBlank()) continue
            val id = vocab.getOrDefault(token, unkId)
            tokens.add(id)
            if (tokens.size >= maxLen - 1) break
        }
        tokens.add(sepId)
        return tokens.toIntArray()
    }
}

5.2 BgeOnDeviceEmbedder（CLS 池化 + L2 归一化）

import ai.onnxruntime.*

class BgeOnDeviceEmbedder(context: Context) {
    private val env = OrtEnvironment.getEnvironment()
    private val session: OrtSession
    private val tokenizer: BertTokenizer
    private val dim = 512

    init {
        val modelBytes = context.assets.open("models/model_qint8.onnx").readBytes()
        val opts = OrtSession.SessionOptions().apply {
            setIntraOpNumThreads(2)  // 铁律：2 线程最稳定
            addConfigEntry("session.intra_op.allow_spinning", "0")
        }
        session = env.createSession(modelBytes, opts)
        tokenizer = BertTokenizer.fromAssets(context)
    }

    fun embed(text: String): FloatArray {
        // BGE 官方查询前缀
        val processedText = "为这个句子生成表示以用于检索相关文章: $text"

        val inputIds = tokenizer.encode(processedText)
        val attentionMask = IntArray(inputIds.size) { 1 }

        val inputTensor = OnnxTensor.createTensor(env, arrayOf(inputIds))
        val maskTensor = OnnxTensor.createTensor(env, arrayOf(attentionMask))

        val results = session.run(mapOf(
            "input_ids" to inputTensor,
            "attention_mask" to maskTensor
        ))

        val outputTensor = results.get("last_hidden_state").get()
        val shape = outputTensor.shape  // [1, seq_len, 512]
        val seqLen = shape[1].toInt()
        val dim = shape[2].toInt()  // 512

        val floatData = outputTensor.getFloatArray()

        // 【核心修正】取 CLS token（位置 0）作为句向量
        val clsEmbedding = FloatArray(dim)
        for (j in 0 until dim) {
            clsEmbedding[j] = floatData[j]
        }

        // 【关键】L2 归一化
        val norm = kotlin.math.sqrt(clsEmbedding.sumOf { it * it })
        return clsEmbedding.map { it / norm }.toFloatArray()
    }

    fun release() {
        session.close()
        env.close()
    }
}

6.1 版本化知识库管理

class VersionedKnowledgeBase(
    private val context: Context,
    private val targetVersion: Int = 2
) {
    private val dbFileName = "knowledge_v${targetVersion}.db"
    private val dbFile = File(context.filesDir, dbFileName)
    private val versionFile = File(context.filesDir, "kb_version.txt")
    private var db: SQLiteDatabase? = null

    suspend fun ensureReady(forceRefresh: Boolean = false) {
        withContext(Dispatchers.IO) {
            NativeLibLoader.load()

            val localVersion = if (versionFile.exists()) {
                versionFile.readText().toIntOrNull() ?: -1
            } else -1

            if (forceRefresh || localVersion != targetVersion || !dbFile.exists()) {
                try {
                    downloadFromCdn("https://your-cdn.com/knowledge_v${targetVersion}.db", dbFile)
                } catch (e: Exception) {
                    context.assets.open(dbFileName).use { input ->
                        dbFile.outputStream().use { output ->
                            input.copyTo(output)
                        }
                    }
                }
                versionFile.writeText(targetVersion.toString())
            }

            db = SQLiteDatabase.openDatabase(dbFile.absolutePath, null, SQLiteDatabase.OPEN_READONLY)
            val cursor = db?.rawQuery("PRAGMA user_version", null)
            cursor?.moveToFirst()
            val dbVersion = cursor?.getInt(0) ?: -1
            cursor?.close()

            if (dbVersion != targetVersion) {
                throw IllegalStateException("数据库版本不匹配: 期望 $targetVersion, 实际 $dbVersion")
            }
        }
    }

    fun search(queryVec: FloatArray, topK: Int = 5): List {
        if (queryVec.size != 512) {
            throw IllegalArgumentException("向量维度必须为 512，当前 ${queryVec.size}")
        }

        val vecBlob = floatArrayToBlob(queryVec)
        val cursor = db?.rawQuery(
            """SELECT ct.text, vec.distance
               FROM kb_vec vec
               JOIN chunk_text ct ON ct.id = vec.rowid
               WHERE vec.embedding MATCH ?
               ORDER BY vec.distance
               LIMIT ?""",
            arrayOf(vecBlob, topK.toString())
        ) ?: return emptyList()

        val results = mutableListOf()
        while (cursor.moveToNext()) {
            results.add(cursor.getString(0))
        }
        cursor.close()
        return results
    }

    private fun floatArrayToBlob(arr: FloatArray): ByteArray {
        val bb = java.nio.ByteBuffer.allocate(arr.size * 4)
        bb.order(java.nio.ByteOrder.LITTLE_ENDIAN)
        arr.forEach { bb.putFloat(it) }
        return bb.array()
    }

    fun close() {
        db?.close()
    }
}

6.2 CDN 下载工具

suspend fun downloadFromCdn(url: String, destFile: File) {
    val client = OkHttpClient.Builder()
        .connectTimeout(30, TimeUnit.SECONDS)
        .readTimeout(60, TimeUnit.SECONDS)
        .build()

    val request = Request.Builder().url(url).build()
    val response = client.newCall(request).await()

    if (!response.isSuccessful) {
        throw IOException("下载失败: ${response.code}")
    }

    response.body?.byteStream()?.use { input ->
        destFile.outputStream().use { output ->
            input.copyTo(output)
        }
    }
}

class CloudGenerator(private val proxyUrl: String) : TextGenerator {
    private val client = OkHttpClient.Builder()
        .connectTimeout(10, TimeUnit.SECONDS)
        .readTimeout(30, TimeUnit.SECONDS)
        .build()

    override suspend fun generate(prompt: String): String {
        val json = JSONObject().apply {
            put("model", "glm-4-flash")
            put("messages", JSONArray().apply {
                put(JSONObject().apply {
                    put("role", "user")
                    put("content", prompt)
                })
            })
            put("stream", false)
        }

        val request = Request.Builder()
            .url(proxyUrl)
            .post(json.toString().toRequestBody("application/json".toMediaType()))
            .build()

        return withContext(Dispatchers.IO) {
            try {
                val response = client.newCall(request).await()
                val body = response.body?.string() ?: ""
                val jsonResp = JSONObject(body)
                jsonResp.getJSONArray("choices")
                    .getJSONObject(0)
                    .getJSONObject("message")
                    .getString("content")
            } catch (e: Exception) {
                "网络异常，请稍后重试"
            }
        }
    }
}

7.3 本地生成器（离线模式，懒加载 400MB）

class LocalLlmGenerator(context: Context) : TextGenerator {
    private var model: LlamaModel? = null
    private val modelFile = File(context.filesDir, "qwen_0.5b_q4_k_m.gguf")
    private var isLoaded = false

    suspend fun loadModel(progress: suspend (Float) -> Unit = {}) {
        withContext(Dispatchers.IO) {
            if (!modelFile.exists()) {
                downloadFile("https://your-cdn.com/qwen_0.5b_q4_k_m.gguf", modelFile, progress)
            }
            model = LlamaModel(modelFile.absolutePath)
            isLoaded = true
        }
    }

    override suspend fun generate(prompt: String): String {
        if (!isLoaded || model == null) {
            return "离线模型未加载，请检查网络后重试"
        }

        // Qwen2.5 必须套 Chat Template
        val formattedPrompt = """
            <|im_start|>system
            你是智能助手，请根据提供的资料回答问题。如果资料中没有相关信息，请如实回答不知道。
            <|im_end|>
            <|im_start|>user
            $prompt
            <|im_end|>
            <|im_start|>assistant
        """.trimIndent()

        return withContext(Dispatchers.IO) {
            try {
                model?.generate(formattedPrompt, maxTokens = 512) ?: "生成失败"
            } catch (e: Exception) {
                "生成失败: ${e.message}"
            }
        }
    }
}

8. 阶段六：RAG 流水线（上下文裁剪 + 超时熔断）

8.1 完整流水线

class RagPipeline(
    private val embedder: BgeOnDeviceEmbedder,
    private val kb: VersionedKnowledgeBase,
) {
    suspend fun query(question: String): String {
        return withTimeoutOrNull(8000) {
            try {
                val vec = withContext(Dispatchers.IO) {
                    embedder.embed(question)
                }

                val chunks = kb.search(vec, topK = 5)
                if (chunks.isEmpty()) {
                    return@withTimeoutOrNull "知识库中未找到相关信息"
                }

                val context = buildTrimmedContext(chunks, maxTokens = 600)

                val prompt = """
                    根据以下资料回答问题。如果资料中没有相关信息，请如实回答"不知道"。

                    资料：
                    $context

                    问题：$question

                    回答：
                """.trimIndent()

                ModelSelector.generate(prompt)

            } catch (e: Exception) {
                "系统繁忙，请稍后再试"
            }
        } ?: "请求超时，请检查网络连接"
    }

    private fun buildTrimmedContext(chunks: List, maxTokens: Int): String {
        var result = ""
        var count = 0
        for (chunk in chunks) {
            val len = chunk.length / 2
            if (count + len > maxTokens) {
                val remain = maxTokens - count
                result += chunk.take(remain * 2) + "......"
                break
            }
            result += chunk + "\n---\n"
            count += len
        }
        return result
    }
}

9. 避坑指南（全量汇总）

坑点现象解决方案
向量维度写错检索结果完全随机或崩溃BGE 维度是 512，不是 384
池化策略错误检索准确率极低（recall < 0.2）BGE 必须用 CLS 池化
忘记 L2 归一化余弦相似度计算结果错误嵌入输出后做 vec / norm(vec)
忘记加查询前缀检索效果比关键词还差加 "为这个句子生成表示以用于检索相关文章: "
Assets 拷贝 ANR首次启动复制文件卡死 UI用 withContext(Dispatchers.IO) 异步拷贝
模拟器加载 .so 失败UnsatisfiedLinkErrorjniLibs/x86_64 下必须放 libsqlite3.so
本地模型生成乱码Qwen 回答重复字符必须套 Qwen Chat Template
知识库版本错乱更新 App 后检索结果全错PC 端写 PRAGMA user_version，App 端校验
API Key 泄露APK 被反编译，Key 被盗刷必须走后端代理（Cloudflare Workers）
ONNX 线程数过多手机发热降频，推理变慢setIntraOpNumThreads(2)
